```js
// record.js
const { chromium } = require('playwright');

(async () => {
  const browser = await chromium.launch({ headless: false });
  const page = await browser.newPage({ viewport: { width: 1920, height: 1080 } });
  await page.goto('file:///C:/Users/Thaió/OneDrive/Documentos/Iwato/IWATO/Biosite%20e%20site%20Maquiadora%20Sue/site_maquiadora_sue.html');
  
  // Hero
  await page.waitForTimeout(2000);
  await page.hover('.btn-luxury-cta');
  await page.waitForTimeout(1000);
  
  // Mobile view
  await page.setViewportSize({ width: 375, height: 812 });
  await page.click('.nav-toggle');
  await page.waitForTimeout(1000);
  await page.click('.nav-toggle');
  
  // Desktop again
  await page.setViewportSize({ width: 1920, height: 1080 });
  
  // Scroll sections
  for (const sel of ['#sobre', '#experiencia', '#servicos', '.portfolio-section', '.testimonials-section', '.action-banner']) {
    await page.locator(sel).scrollIntoViewIfNeeded();
    await page.waitForTimeout(1500);
  }
  
  // Lightbox
  await page.locator('.portfolio-item').first().click();
  await page.waitForTimeout(1000);
  await page.keyboard.press('ArrowRight');
  await page.waitForTimeout(800);
  await page.keyboard.press('Escape');
  
  await browser.close();
})();
```
